public class Rectangulo{
            //Atributos
            int alto;
            int ancho;
            boolean relleno;
            // Métodos
            public int devolverArea(){
                return alto * ancho;
            }
            public void rellenar(boolean r){
                relleno = r;
            }
            public void cambiarTamaño(int an, int al){
                ancho = an;
                alto = al;
            }
            public Rectangulo(){ // Constructor
                alto = 10;
                ancho = 10;
                relleno = false;
            }
        } // Fin de la clase rectangulo
